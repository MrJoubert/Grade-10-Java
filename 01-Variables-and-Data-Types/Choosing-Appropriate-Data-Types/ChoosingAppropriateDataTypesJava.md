/*
 * Topic: Choosing Appropriate Data Types
 * Demonstrates suitable data types for school information.
 */

public class ChoosingAppropriateDataTypes {

    public static void main(String[] args) {

        // Store the Student's information.
        String studentName = "Olivia Smith";
        String studentNumber = "001307";
        int studentAge = 17;
        int studentGrade = 11;
        double averageMark = 84.25;
        char studentInitial = 'O';
        boolean playsSport = false;
        String subjectName = "Information Technology";

        // Display the Student's profile.
        System.out.println("SCHOOL STUDENT PROFILE");
        System.out.println("======================");

        System.out.println("Name: " + studentName);
        System.out.println("Student number: " + studentNumber);
        System.out.println("Initial: " + studentInitial);
        System.out.println("Age: " + studentAge);
        System.out.println("Grade: " + studentGrade);
        System.out.println("Average mark: " + averageMark);
        System.out.println("Participates in sport: " + playsSport);
        System.out.println("Subject: " + subjectName);

    }

}
